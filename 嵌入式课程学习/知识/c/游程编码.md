/* rle 游程编码（以前写的rlc是错误名字）游程编码和解码 重复字节压缩和解压算法  */

#include <corecrt.h>
#include <stdlib.h>
#include <stdio.h>


/* rle编码 */
/* 传入参数: 源数据空间 + 长度 */
/* 返回值: 编码数据空间 + 长度(通过参数返回) */
/**
 * @brief rle编码
 * 
 * @param src_data 被编码的数据空间
 * @param len 传入位B，需要传入被编码数据空间的大小；同时也会通过len返回编码后的空间大小
 * @return void* 返回编码后的空间
 */
void *Rle_Encode(const void *src_data, size_t *len){
    /* 1. 检查数据有效性,以及临界条件判断 */
    if(src_data == NULL || len == NULL) { return NULL; }
    if(*len == 0){ return NULL; }   /* 源空间没有数据的情况 */

    /* 2. 统计编码空间需要申请的空间大小 */
    const unsigned char *data = (const unsigned char *)src_data;
    int count = 0;
    size_t src_len = *len;
    int result_len = 0;
    for (int i = 0; i < src_len; ++i) {
        ++count;
        if ( i == src_len - 1 || data[i] != data[i + 1] || count == 255) {
            result_len += 2;
        }
    }
    // printf("%d\n", result_len);

    /* 3. 申请空间 */
    unsigned char *result_data = (unsigned char *)malloc(sizeof(unsigned char) * result_len);
    if(result_data == NULL) { return NULL;}

    /* 4. 填充空间 */
    /* 将源数据空间中的数据一组一组的处理 */
    count = 0;
    int result_index = 0;
    for(int i = 0; i < src_len; ++i){
        ++count;
        if(i == src_len - 1 || data[i] != data[i + 1] || count == 255){
            result_data[result_index++] = data[i];
            result_data[result_index++] = count;
            count = 0;
        }
    }

    /* 5. 返回编码空间 */
    *len = result_len;
    return result_data;
}

/* rle解码 */
/* 输入参数：被解码数据 + 长度 */
/* 返回值：解码空间 + 长度 */
/**
 * @brief rle解码
 * 
 * @param src_data 被解码的数据空间
 * @param len 传入位B，需要传入被解码数据空间的大小；同时也会通过len返回解码后的空间大小
 * @return void* 返回解码后的空间
 */
void * Rle_Decode(const  void *src_data, size_t *len){
    /* 1. 数据有效性 */
    if( src_data == NULL || len == NULL ){ return NULL; }
    if(*len == 0) { return NULL; }

    /* 2. 解码空间的大小 */
    const unsigned char *data = (const unsigned char *)src_data;
    size_t src_len = *len;
    size_t result_len = 0;
    for(int i = 1; i < src_len; i += 2){
        result_len += data[i];
    }
    // printf("%zd \n", result_len);

    /* 3. 申请解码空间 */
    unsigned char *result_data = (unsigned char *)malloc(sizeof(unsigned char) * result_len);
    if(result_data == NULL) { return NULL; }

    /* 4. 填充解码空间 */
    int result_index = 0;
    for (int i = 0; i < src_len; i += 2) {
        for(int j = 0; j < data[i + 1]; ++j){
            result_data[result_index++] = data[i];
        }
    }

    /* 5. 返回解码空间 */
    *len = result_len;
    return result_data;
}

/* 释放资源 */
/**
 * @brief rle空间释放
 * 
 * @param data 传入需要销毁的空间
 */
void Rle_release(void *data){
    if(data != NULL){
        free(data);
    }
}

/* 联合测试: 编码后直接解码，比对前后数据是否完全一致 */
void test_comprehensive_loop() {
    printf("--- Start Comprehensive Loop Test ---\n");
    
    /* 准备一个复杂的源数据，包含超长重复、单字节、0x00和0xFF */
    size_t src_len = 400;
    char *src_data = (char *)malloc(src_len);
    
    int idx = 0;
    // 1. 260个0xFF (触发无符号255边界)
    for(int i = 0; i < 260; ++i) src_data[idx++] = (char)0xFF;
    // 2. 3个0x00
    for(int i = 0; i < 3; ++i)   src_data[idx++] = 0x00;
    // 3. 120个0x55
    for(int i = 0; i < 120; ++i) src_data[idx++] = 0x55;
    // 4. 17个不重复的渐变数据
    for(int i = 0; i < 17; ++i)  src_data[idx++] = (char)i;

    /* 第一步：编码 */
    size_t enc_len = src_len;
    char *enc_data = (char *)Rle_Encode(src_data, &enc_len);
    if (enc_data == NULL) {
        printf("Loop Test: Encode Failed!\n");
        free(src_data);
        return;
    }
    printf("Encoded size from %zu to %zu\n", src_len, enc_len);

    /* 第二步：解码 */
    size_t dec_len = enc_len;
    char *dec_data = (char *)Rle_Decode(enc_data, &dec_len);
    if (dec_data == NULL) {
        printf("Loop Test: Decode Failed! (Check if signed char caused negative count)\n");
        Rle_release(enc_data);
        free(src_data);
        return;
    }

    /* 第三步：校验 */
    int match = 1;
    if (dec_len != src_len) {
        printf("Dimension mismatch! Src: %zu, Dec: %zu\n", src_len, dec_len);
        match = 0;
    } else {
        for (size_t i = 0; i < src_len; ++i) {
            if (src_data[i] != dec_data[i]) {
                printf("Data mismatch at index %zu! Src: 0x%02x, Dec: 0x%02x\n", 
                        i, (unsigned char)src_data[i], (unsigned char)dec_data[i]);
                match = 0;
                break;
            }
        }
    }

    if (match) {
        printf("SUCCESS: Perfect match after Encode & Decode loop!\n");
    } else {
        printf("FAILED: Data corruption detected.\n");
    }

    /* 资源释放 */
    Rle_release(dec_data);
    Rle_release(enc_data);
    free(src_data);
}

/* 总和测试2: 非char复杂结构化数据测试（结构体/多字节数组） */
void test_comprehensive_non_char() {
    printf("--- Start Comprehensive Non-Char Test ---\n");

    /* 1. 定义一个模拟的像素/点信息结构体 */
    typedef struct {
        unsigned short x; /* 2 字节 */
        unsigned char r;  /* 1 字节 */
        unsigned char g;  /* 1 字节 */
        unsigned char b;  /* 1 字节 */
    } Point; /* 该结构体在大多数系统上由于对齐会占用 6 或 8 字节 */

    /* 2. 构造一个包含大量连续重复“点数据”的复杂数组 */
    size_t num_elements = 100;
    size_t src_len = num_elements * sizeof(Point);
    Point *src_data = (Point *)malloc(src_len);

    Point heavy_point = { 50000, 0xFF, 0x00, 0xAA }; /* 带有大整数和边界字节的点 */
    Point light_point = { 12,    0x11, 0x22, 0x33 };

    int idx = 0;
    // 放入 60 个完全相同的 heavy_point（转换为字节流后会产生规律的交替重复）
    for(int i = 0; i < 60; ++i) {
        src_data[idx++] = heavy_point;
    }
    // 放入 40 个完全相同的 light_point
    for(int i = 0; i < 40; ++i) {
        src_data[idx++] = light_point;
    }

    /* 3. 第一步：将结构体数据作为纯粹的字节流进行 RLE 编码 */
    size_t enc_len = src_len;
    unsigned char *enc_data = (unsigned char *)Rle_Encode(src_data, &enc_len);
    if (enc_data == NULL) {
        printf("Non-Char Test: Encode Failed!\n");
        free(src_data);
        return;
    }
    printf("Point Array compressed from %zu bytes to %zu bytes\n", src_len, enc_len);

    /* 4. 第二步：解码恢复字节流 */
    size_t dec_len = enc_len;
    Point *dec_data = (Point *)Rle_Decode(enc_data, &dec_len);
    if (dec_data == NULL) {
        printf("Non-Char Test: Decode Failed!\n");
        Rle_release(enc_data);
        free(src_data);
        return;
    }

    /* 5. 第三步：从结构体和内存双重维度进行严格校验 */
    int match = 1;
    if (dec_len != src_len) {
        printf("Dimension mismatch! Src: %zu bytes, Dec: %zu bytes\n", src_len, dec_len);
        match = 0;
    } else {
        // 强转回结构体指针进行业务逻辑比对
        for (size_t i = 0; i < num_elements; ++i) {
            if (src_data[i].x != dec_data[i].x || 
                src_data[i].r != dec_data[i].r || 
                src_data[i].g != dec_data[i].g || 
                src_data[i].b != dec_data[i].b) {
                printf("Structure mismatch at element %zu!\n", i);
                match = 0;
                break;
            }
        }
    }

    if (match) {
        printf("SUCCESS: Non-Char structured data loops back perfectly!\n");
    } else {
        printf("FAILED: Structured data corruption detected.\n");
    }

    /* 6. 资源释放 */
    Rle_release(dec_data);
    Rle_release(enc_data);
    free(src_data);
}
